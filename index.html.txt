<script>
    // Cargar datos de LocalStorage o usar predeterminados
    let db = JSON.parse(localStorage.getItem('studio21_db')) || [
        { n: "Sicla", t: "9241287666", f: "S21-2001", h: [], g: "F" },
        { n: "Alan", t: "8119871595", f: "S21-2004", h: [], g: "M" }
    ];

    let actual = null;

    function saveDB() {
        localStorage.setItem('studio21_db', JSON.stringify(db));
    }

    function resetFecha() {
        const ahora = new Date();
        ahora.setMinutes(ahora.getMinutes() - ahora.getTimezoneOffset());
        const stringFecha = ahora.toISOString().slice(0, 16);
        document.getElementById('sFecha').value = stringFecha;
    }

    function buscar() {
        let q = document.getElementById('findInput').value.toLowerCase().trim();
        if(!q) return;
        
        let c = db.find(x => x.n.toLowerCase().indexOf(q) !== -1 || x.f.toLowerCase().indexOf(q) !== -1);
        
        if(c) {
            actual = c;
            showView('profile');
            document.getElementById('dN').innerText = c.n;
            document.getElementById('dF').innerText = c.f;
            resetFecha();
            updateWA();
            renderH();
        } else { alert("No se encontró el cliente."); }
    }

    function registrar() {
        let n = document.getElementById('regN').value;
        let t = document.getElementById('regT').value;
        let g = document.getElementById('regG').value;
        
        if(n && t) {
            let f = "S21-" + (db.length + 2001);
            db.push({ n: n, t: t, f: f, h: [], g: g });
            saveDB();
            alert("Cliente registrado: " + f);
            document.getElementById('regN').value = "";
            document.getElementById('regT').value = "";
            buscarDirecto(f);
        } else { alert("Completa nombre y teléfono."); }
    }

    function buscarDirecto(folio) {
        document.getElementById('findInput').value = folio;
        buscar();
    }

    function guardarV() {
        let s = document.getElementById('sTipo').value;
        let n = document.getElementById('sNotas').value;
        let f = document.getElementById('sFecha').value.replace('T', ' ');
        
        actual.h.unshift({ fecha: f, servicio: s, nota: n });
        saveDB();
        renderH();
        document.getElementById('sNotas').value = "";
        alert("¡Visita guardada!");
    }

    function renderH() {
        let w = document.getElementById('hWall');
        if(actual.h.length === 0) {
            w.innerHTML = "<p style='color:#444'>Sin visitas registradas.</p>";
            return;
        }
        w.innerHTML = actual.h.map(x => `
            <div class="history-card">
                <small style="color:var(--oro)">${x.fecha}</small><br>
                <strong>${x.servicio}</strong>
                <p style="margin:5px 0; font-size:12px; color:#aaa;">${x.nota || ''}</p>
            </div>
        `).join('');
    }

    function updateWA() {
        const msj = encodeURIComponent("¡Hola " + actual.n + "! Te saludamos de STUDIO 21. ✨");
        document.getElementById('waLink').href = "https://wa.me/" + actual.t + "?text=" + msj;
    }

    function showView(viewId) {
        document.getElementById('home').classList.remove('active');
        document.getElementById('home').style.display = 'none';
        document.getElementById('profile').classList.remove('active');
        document.getElementById('profile').style.display = 'none';
        
        const activeView = document.getElementById(viewId);
        activeView.style.display = 'block';
        setTimeout(() => { activeView.classList.add('active'); }, 10);
    }

    function home() {
        showView('home');
        document.getElementById('findInput').value = "";
    }
</script>
